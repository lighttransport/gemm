cmake_minimum_required(VERSION 3.21)
project(vhuman_sensevoice LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SENSEVOICE_SOURCE "${REPO_ROOT}/tmp/vhuman-emotion/source" CACHE PATH "Pinned FunASR source")
set(LLAMA_SOURCE "${REPO_ROOT}/tmp/vhuman-emotion/llama" CACHE PATH "Pinned ggml source")
execute_process(COMMAND git -C "${LLAMA_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT LLAMA_REV STREQUAL "803b7fcae893e9caaee3921779628fef83ac0965")
    message(FATAL_ERROR "ggml source does not match the pinned release dependency")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_source.py" "${SENSEVOICE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/funasr-sensevoice.cpp" COMMAND_ERROR_IS_FATAL ANY)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_HIP ON CACHE BOOL "Build native ROCm kernels")
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}/ggml" ggml)
add_executable(llama-funasr-sensevoice "${CMAKE_CURRENT_BINARY_DIR}/funasr-sensevoice.cpp")
target_include_directories(llama-funasr-sensevoice PRIVATE "${SENSEVOICE_SOURCE}/runtime/llama.cpp/funasr-common")
target_link_libraries(llama-funasr-sensevoice PRIVATE ggml)
set_target_properties(llama-funasr-sensevoice PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
