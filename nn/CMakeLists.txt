cmake_minimum_required(VERSION 3.16)
project(gemm_nn LANGUAGES C)
include(CTest)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/gn_kernels.inc
  COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/embed_kernels.py
          ${CMAKE_CURRENT_SOURCE_DIR}/gn_kernels.cu ${CMAKE_CURRENT_BINARY_DIR}/gn_kernels.inc
  DEPENDS gn_kernels.cu gn_sm120.cuh gn_rdna4.cuh embed_kernels.py VERBATIM)
add_library(gn STATIC gn.c gn_cpu.c gn_gpu.c ../cuda/cuew.c ../rdna4/rocew.c
  ${CMAKE_CURRENT_BINARY_DIR}/gn_kernels.inc)
# Loader pointer variables have the vendor function names. Keep them local,
# including when this archive is linked into an executable with -rdynamic.
set_source_files_properties(../cuda/cuew.c ../rdna4/rocew.c
  PROPERTIES COMPILE_OPTIONS "-fvisibility=hidden")
set_target_properties(gn PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON)
target_include_directories(gn PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}
  PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(gn PUBLIC Threads::Threads m ${CMAKE_DL_LIBS})
target_compile_definitions(gn PRIVATE _POSIX_C_SOURCE=200809L)
option(GN_HIPBLASLT "Enable optional AMD hipBLASLt backend (requires ROCm SDK)" OFF)
option(GN_GPU_TESTS "Run hardware tests on RX 9070 XT (requires GN_HIPBLASLT)" OFF)
if(GN_GPU_TESTS AND NOT GN_HIPBLASLT)
  message(FATAL_ERROR "GN_GPU_TESTS requires GN_HIPBLASLT=ON")
endif()
if(GN_HIPBLASLT)
  enable_language(CXX)
  find_path(HIPBLASLT_INCLUDE_DIR hipblaslt/hipblaslt.h HINTS /opt/rocm/core/include /opt/rocm/include)
  find_path(HIP_INCLUDE_DIR hip/hip_runtime.h HINTS /opt/rocm/core/include /opt/rocm/include)
  find_library(HIPBLASLT_LIBRARY hipblaslt HINTS /opt/rocm/core/lib /opt/rocm/lib)
  find_library(HIP_RUNTIME_LIBRARY amdhip64 HINTS /opt/rocm/core/lib /opt/rocm/lib)
  if(NOT HIPBLASLT_INCLUDE_DIR OR NOT HIP_INCLUDE_DIR OR NOT HIPBLASLT_LIBRARY OR NOT HIP_RUNTIME_LIBRARY)
    message(FATAL_ERROR "GN_HIPBLASLT requires the hipBLASLt and HIP development headers/libraries")
  endif()
  target_sources(gn PRIVATE gn_hipblaslt.cpp)
  target_include_directories(gn PRIVATE ${HIPBLASLT_INCLUDE_DIR} ${HIP_INCLUDE_DIR})
  target_compile_definitions(gn PRIVATE GN_HIPBLASLT=1 $<$<COMPILE_LANGUAGE:CXX>:__HIP_PLATFORM_AMD__>)
  set_target_properties(gn PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
  target_link_libraries(gn PUBLIC ${HIPBLASLT_LIBRARY} ${HIP_RUNTIME_LIBRARY})
endif()
add_executable(gn_tool gn_tool.c)
target_link_libraries(gn_tool PRIVATE gn)
if(BUILD_TESTING)
  add_executable(gn-test test_gn.c)
  target_link_libraries(gn-test PRIVATE gn)
  add_test(NAME gn-native COMMAND gn-test ${CMAKE_CURRENT_BINARY_DIR}/roundtrip.safetensors)
  add_executable(gn-checkpoint-io-test test_checkpoint_io.c)
  target_link_libraries(gn-checkpoint-io-test PRIVATE gn)
  add_test(NAME gn-checkpoint-io COMMAND gn-checkpoint-io-test
    ${CMAKE_CURRENT_BINARY_DIR}/checkpoint-io.safetensors)
  add_executable(gn-gpu-test test_gpu.c)
  target_link_libraries(gn-gpu-test PRIVATE gn)
  set_target_properties(gn-gpu-test PROPERTIES ENABLE_EXPORTS ON)
  add_test(NAME gn-loader-symbols COMMAND ${Python3_EXECUTABLE} -B
    ${CMAKE_CURRENT_SOURCE_DIR}/test_loader_symbols.py $<TARGET_FILE:gn-gpu-test>)
  add_executable(gn-gpu-graph-test test_gpu_graph.c)
  target_link_libraries(gn-gpu-graph-test PRIVATE gn)
  if(GN_GPU_TESTS)
    add_test(NAME gn-gpu-graph COMMAND gn-gpu-graph-test)
    set_tests_properties(gn-gpu-graph PROPERTIES TIMEOUT 120)
  endif()
endif()
