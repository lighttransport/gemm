 about the loop body:
	//   B_row = ld1w (p0) [x20]   // one SVE vector of B
	//   for m = 0..7:
	//       a = A[m][k] (scalar)
	//       z[m] = fmla (z[m], p0/m, a, B_row)
	//   A += 8 (skip 8 floats = 1 per row)
	//   B += VL (skip VL floats = 1 row of B)
	//   K -= 1

	// Load B row (one SVE vector of floats).
	ld1w	z16.s, p0/z, [x20]

	// Load 8 A scalars (one per row) from contiguous memory.
	// A layout: row0[k], row1[k], ..., row7[k] (8 floats contiguous).
	ldr	q17, [x19]		// 128-bit load = 4 floats (rows 0-3)
	ldr	q18, [x19, #16]		// next 4 floats (rows 4-7)

	// Broadcast each scalar and FMA.
	// Use dup on each element of the Q registers.
	// For rows 0-3: q17 = [a0, a1, a2, a3] (little-endian: a0 at lowest addr)
	// For rows 4-7: q18 = [a4, a5, a6, a7]
	// We'll dup each lane into a full SVE vector and fmla.

	// Row 0: a0 = q17.s[0]
	dup	z20.s, z17.s[0]
	fmla	z0.s, p0/m, z20.s, z16.s

	// Row 1: a1 = q17.s[1]
	dup	z20.s, z17.s[1]
	fmla	z1.s, p0/m, z20.s, z16.s

	// Row 2: a2 = q17.s[2]
	dup	z20.s, z17.s[2]
	fmla	z2.s, p0/m, z20.s, z16.s

	// Row 3: a3 = q17.s[3]
	dup	z20.s, z17.s[3]
	fmla	z3.s, p0/m, z20.s, z16.s

	// Row 4: a4 = q18.s[0]
	dup	z20.s, z18.s[0]
	fmla	z4.s, p0/m, z20.s, z16.s

	// Row 5: a5 = q18.s[1]
	dup	z20.s, z18.s[1]
	fmla	z5.s, p0/m, z20.s, z16.s

	// Row 6: a6 = q18.s[2]
	dup	z20.s, z18.s[2]
	fmla	z6.s, p0/m, z20.s, z16.s

	// Row 7: a7 = q18.s[3]
	dup	z20.s, z18.s[3]
	fmla	z7.s, p0/m, z20.s, z16.s

	// Advance pointers.
	add	x19, x19, #32		// A += 8 floats (8 rows * 1 element)
	add	x20, x20, x22, lsl #2	// B += VL floats (1 row of B)
	// Wait: x22 is ldc, not VL. We need VL in bytes. Use cntw.
	// Actually B stride is VL floats = cntw * 4 bytes.
	// Simpler: add x20, x20, x22, lsl #2 is WRONG. We need to add VL*4.
	// Use cntw to get VL in elements, multiply by 4.
	// But cntw is a pseudo-instruction for rdvl (or we can use rdvl).
	// Let's compute VL in bytes: cntw x24, then add x20, x20, x24, lsl #2.
	// But cntw writes to a register. We'll use x24.
	// However we already used x24? No, x24 is free (callee-saved but we saved it).
	// Let's insert after the loop body.
	// Actually we can compute VL once before the loop and store in a callee-saved.
	// Simpler: use rdvl x24, #1 (returns VL in bytes for SVE).
	// Then add x20, x20, x24.
	// But rdvl is an SVE instruction. We'll do:
	//   rdvl	x24, #1		// x24 = VL in bytes
	//   add	x20, x20, x24
	// However we need to do this every iteration. That's fine.
	// But we also need to decrement K.
	sub	x23, x23, #1
	cbnz	x23, 1b

	// After loop, store the 8 accumulator vectors to C.
	// C[m] = z[m] for m=0..7, with row stride ldc (in elements).
	// Compute VL in bytes once more (or reuse from loop).
	rdvl	x24, #1			// VL in bytes
	// For each row m:
	//   st1w {z[m].s}, p0, [C_base + m*ldc*4]
	// C_base = x21, ldc = x22 (elements), so byte offset = m * (ldc * 4).
	// We'll compute base pointers for each row.
	// Use x9..x16 as scratch (we saved x19-x24, so x9-x18 are free).
	mov	x9, x21			// C row 0 base
	st1w	z0.s, p0, [x9]

	add	x10, x9, x22, lsl #2	// C row 1 = base + ldc*4
	st1w	z1.s, p0, [x10]

	add	x11, x10, x22, lsl #2	// row 2
	st1w	z2.s, p0, [x11]

	add	x12, x11, x22, lsl #2	// row 3
	st1w	z3.s, p0, [x12]

	add	x13, x12, x22, lsl #2	// row 4
	st1w	z4.s, p0, [x13]

	add	x14, x13, x22, lsl #2	// row 5
	st1w	z5.s, p0, [x14]

	add	x15, x14, x22, lsl #2	// row 6
	st1w	z6.s, p0, [x15]

	add	x16
